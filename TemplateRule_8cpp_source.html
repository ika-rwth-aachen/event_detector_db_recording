52void TemplateRule::evaluate() {
62 // create empty EvaluationResult
63 EvaluationResult result;
64
65 // fetch list of clients with point clouds in the buffer
66 std::vector<std::string> lidar_clients = buffer_->sensor_msgs__PointCloud2.getClients();
67 if (lidar_clients.empty()) return;
68
69 // fetch latest pose from the first identified client
70 std::vector<ed::Stamped<geometry_msgs::msg::PoseStamped>> stamped_pose =
71 buffer_->geometry_msgs__PoseStamped.get(1, lidar_clients[0], "");
72 if (stamped_pose.empty()) return;
81 // compute current distance of client to origin of observed area
82 double dx = stamped_pose[0].msg->pose.position.x - x_;
83 double dy = stamped_pose[0].msg->pose.position.x - y_;
84 double r = std::sqrt(std::pow(dx, 2) + std::pow(dy, 2));
85
86 // in this case, data is considered to be relevant when the client position
87 // lies within a specified radius (the observed area)
88 if (r <= radius_) {
89 // create a ClientDataRequest for the client including information about
90 // what types of data from what time frame to request for long-term storage
91 ClientDataRequest v_request;
92 v_request.start_time = stamped_pose[0].msg->header.stamp;
93 v_request.end_time = stamped_pose[0].msg->header.stamp;
94 // v_request.tf2_msgs__TFMessage.request = true; // transforms requested by default
95 v_request.geometry_msgs__PoseStamped.request = true;
96 v_request.geometry_msgs__PoseStamped.n_samples = 1;
97 v_request.sensor_msgs__PointCloud2.request = true;
98 v_request.sensor_msgs__PointCloud2.n_samples = 1;
99 // Note that what the timeframe refers to depends on the buffer
100 // configuration (see comment above)
101
102 // move the ClientDataRequest to an encompassing ScenarioDataRequest (which
103 // may contain requests for multiple clients)
104 ScenarioDataRequest s_request;
105 s_request.client_requests[lidar_clients[0]] = v_request;
106
107 // assign scenario types and optionally add arbitrarily nested JSON metadata
108 s_request.scenario_types = scenario_types_;
109 std::stringstream ss;
110 ss << "{\"distance\": " << r << "}";
111 s_request.metadata = ss.str();
112
113 // move the ScenarioDataRequest to the EvaluationResult (which may contain
114 // requests for multiple scenarios)
115 result.scenario_requests.push_back(s_request);
116
117 // log info
118 RCLCPP_INFO(rclcpp::get_logger(this->getRuleName()),
119 "[%s] Found relevant data: Client '%s' is %.3fm from observed area "
120 "origin",
121 this->getRuleName().c_str(), lidar_clients[0].c_str(), r);
122
123 // trigger storage of data in database
124 this->trigger(result);
125 }
126}
