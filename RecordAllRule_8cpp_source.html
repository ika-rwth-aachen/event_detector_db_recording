34void RecordAllRule::evaluate() {
35 EvaluationResult result;
36 ScenarioDataRequest s_request;
37 s_request.scenario_types = scenario_types_;
38
39 // create client requests based on available clients per selected datatype
40#define DATATYPE(TYPE, LARGE_TYPE, VAR, COLLECTION) \
41 if (VAR##_) { \
42 std::vector<std::string> VAR##_clients = buffer_->VAR.getClients(); \
43 for (const std::string& client : VAR##_clients) { \
44 if (s_request.client_requests.count(client) == 0) { \
45 s_request.client_requests[client] = ClientDataRequest(); \
46 s_request.client_requests[client].tf2_msgs__TFMessage.request = false; \
47 } \
48 s_request.client_requests[client].VAR.request = true; \
49 } \
50 }
51#include <event_detector_db_recording_plugin/datatypes.macro>
52#undef DATATYPE
53
54 // set current time window to record
55 rclcpp::Time end_time = ed_->now();
56 for (auto& v_request : s_request.client_requests) {
57 v_request.second.start_time = start_time_;
58 v_request.second.end_time = end_time;
59 }
60
61 // log info
62 if (s_request.client_requests.size() > 0) {
63 result.scenario_requests.push_back(s_request);
64 RCLCPP_INFO(rclcpp::get_logger(this->getRuleName()), "Recording %.1fs of selected data from %ld clients",
65 (end_time - start_time_).seconds(), s_request.client_requests.size());
66 }
67
68 // store current end time as starting point for next storage period
69 start_time_ = end_time;
70
71 this->trigger(result);
72}
