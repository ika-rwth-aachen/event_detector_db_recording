32template <typename T>
33ed::string_map<std::vector<ed::Stamped<T>>> RecordingRule::requestClientData(
34 ed::BufferManager::DataBuffers<T>& data_buffer, const DataRequest& data_request, const std::string& client_id,
35 const rclcpp::Time& start_time, const rclcpp::Time& end_time) {
36 ed::string_map<std::vector<ed::Stamped<T>>> requested_samples_by_topic;
37
38 // check if data type is requested
39 if (data_request.request) {
40 int n_samples = data_request.n_samples;
41 std::set<std::string> frames = data_request.frames;
42 if (frames.empty()) frames = {""}; // fetch all, if no frame specified
43
44 // loop over requested frames
45 for (const auto& frame_id : frames) {
46 // fetch data from buffer
47 auto samples_by_topic = data_buffer.getWithTopic(-1, start_time, end_time, client_id, frame_id);
48
49 // loop over topics
50 for (const auto& kv : samples_by_topic) {
51 const std::string& topic = kv.first;
52 auto& samples = kv.second;
53
54 // cap at specified number of samples
55 int n = std::min(n_samples, static_cast<int>(samples.size()));
56 if (n < 0) n = samples.size();
57 int offset = samples.size() - n;
58 for (size_t i = offset; i < samples.size(); i++) requested_samples_by_topic[topic].push_back(samples[i]);
59 }
60 }
61 }
62
63 return requested_samples_by_topic;
64}
