104void RecordingRule::loadRuleParameters() {
105 this->loadRuleParameter<std::string>("database.name", db_config_.name, "db");
106 this->loadRuleParameter<std::string>("database.host", db_config_.host, "localhost");
107 this->loadRuleParameter<int>("database.port", db_config_.port, 27017);
108 this->loadRuleParameter<std::string>("database.user", db_config_.user, "");
109 this->loadRuleParameter<std::string>("database.pass", db_config_.pass, "");
110 std::string large_data_root;
111 this->loadRuleParameter<std::string>("database.large_data_root", large_data_root, "");
112 this->loadRuleParameter<bool>("database.dry_run", db_config_.dry_run, false);
113
114 // process filepaths
115 db_config_.large_data_root = std::filesystem::path(large_data_root);
116 if (!db_config_.large_data_root.has_root_path()) {
117 auto tmp = std::filesystem::path(__FILE__).parent_path().parent_path();
118 tmp.append(large_data_root);
119 db_config_.large_data_root = tmp;
120 }
121 RCLCPP_DEBUG(rclcpp::get_logger(this->getRuleName()), "db_config_.large_data_root = %s", db_config_.large_data_root.c_str());
122
123 // set up DatabaseInterface
124 db_ = std::make_shared<DatabaseInterface>(db_config_.dry_run);
125
126 // set up connection to database
127 db_->connectToDatabase(db_config_.name, db_config_.host, db_config_.port, db_config_.user, db_config_.pass);
128 db_->setLargeDataRoot(db_config_.large_data_root);
129
130 // register connected clients in database
131 const auto connected_clients = ed_->getConnectedClients();
132 for (auto& client : connected_clients) {
133 db_->createConnectedClient(client.id, client.name);
134 db_->registerClientBaseFrame(client.id, client.base_frame);
135 RCLCPP_INFO(rclcpp::get_logger(this->getRuleName()), "Registered client '%s' with ID '%s'", client.name.c_str(),
136 client.id.c_str());
137 }
138}

140void RecordingRule::trigger(const EvaluationResult& result) {
141 // postprocess scenario requests
142 std::vector<ScenarioDataRequest> scenario_requests;
143 for (const auto& scenario_request : result.scenario_requests) {
144 ScenarioDataRequest request = scenario_request;
145
146 // add reponsible rule
147 request.rules = {this->getRuleName().c_str()};
148
149 // check if metadata is parseable
150 try {
151 nlohmann::json j_tmp = nlohmann::json::parse(request.metadata);
152 } catch (nlohmann::json::parse_error& e) {
153 RCLCPP_WARN(rclcpp::get_logger(this->getRuleName()),
154 "Failed to parse scenario metadata as json, will be "
155 "ignored:%s\n'%s'",
156 e.what(), request.metadata.c_str());
157 request.metadata = "{}";
158 }
159
160 scenario_requests.push_back(request);
161 }
162
163 // request and store data
164 for (const auto& request : scenario_requests) this->requestAndStoreInDatabase(request);
165}

167void RecordingRule::requestAndStoreInDatabase(const ScenarioDataRequest& request) {
168 ScenarioData data;
169 data.scenario_types = request.scenario_types;
170 data.rules = request.rules;
171 data.metadata = request.metadata;
172
173 bool request_successful = true;
174
175 // loop over all clients for which data is requested
176 for (const auto& ix : request.client_requests) {
177 std::string client_id = ix.first;
178 const ClientDataRequest& v_request = ix.second;
179
180 // fetch data from buffer
181#define DATATYPE(TYPE, LARGE_TYPE, VAR, COLLECTION) \
182 \
183 /* fetch requested TYPE */ \
184 auto VAR##_by_topic = \
185 this->requestClientData(buffer_->VAR, v_request.VAR, client_id, v_request.start_time, v_request.end_time); \
186 if (!VAR##_by_topic.empty()) { \
187 for (const auto& kv : VAR##_by_topic) { \
188 const auto& samples = kv.second; \
189 auto& data_samples = data.VAR[client_id][kv.first]; \
190 data_samples.insert(data_samples.end(), samples.begin(), samples.end()); \
191 } \
192 } else if (v_request.VAR.request) { \
193 request_successful = false; \
194 RCLCPP_WARN(rclcpp::get_logger(this->getRuleName()), "Failed to request %s from client '%s' in time range (%f, %f)", \
195 #COLLECTION, client_id.c_str(), v_request.start_time.seconds(), v_request.end_time.seconds()); \
196 }
197#include <event_detector_db_recording_plugin/datatypes.macro>
198#undef DATATYPE
199 }
200
201 if (!request_successful) {
202 std::stringstream ss;
203 for (const std::string& rule : data.rules) ss << "'" << rule << "'" << ", ";
204 RCLCPP_WARN(rclcpp::get_logger(this->getRuleName()),
205 "Failed to fetch requested data from buffer, will not store scenario. "
206 "Data was requested by the following rules: %s",
207 ss.str().c_str());
208 return;
209 }
210
211 // store data in database
212 if (!request.client_requests.empty()) db_->storeScenarioData(data);
213}
